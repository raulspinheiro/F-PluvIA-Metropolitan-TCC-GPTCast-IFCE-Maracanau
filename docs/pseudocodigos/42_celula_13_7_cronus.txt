FUNÇÃO _carregar_2025(pasta)

FUNÇÃO _carregar_2026(pasta)

FUNÇÃO _contagens(o, p, thr)
    (ob, pb) ← (o ≥ thr, p ≥ thr)
    eixo ← tuple(range(1, o.ndim))
    RETORNAR np.stack([np.sum(ob & pb, axis=eixo), np.sum(~ob & pb, axis=eixo), np.sum(ob & ~pb, axis=eixo)], −1).astype(float)
FIM FUNÇÃO
FUNÇÃO _f1(c)
    (tp, fp, fn) ← (c[(Ellipsis, 0)], c[(Ellipsis, 1)], c[(Ellipsis, 2)])
    USANDO np.errstate(invalid="ignore", divide="ignore")
        RETORNAR 100 × 2 × tp / 2 × tp + fp + fn
    FIM USANDO
FIM FUNÇÃO
FUNÇÃO _csi(c)
    (tp, fp, fn) ← (c[(Ellipsis, 0)], c[(Ellipsis, 1)], c[(Ellipsis, 2)])
    USANDO np.errstate(invalid="ignore", divide="ignore")
        RETORNAR 100 × tp / tp + fp + fn
    FIM USANDO
FIM FUNÇÃO
